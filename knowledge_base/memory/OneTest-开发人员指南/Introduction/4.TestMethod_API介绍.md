## 4.1.Evaluate API

|  |  |
| --- | --- |
|  | 4.1.Evaluate API |

4.1.Evaluate API

* Evaluate API可用于PTD、FTD。
* 这个API会对每个Site通过比较测量的值和设置的limit值来返回pass/fail结果。
  将测量值与指定的limit值进行比较。 如果结果值介于上下阈值之间，则视为通过。 反之，则认为失败。
* Evaluate API 有两个重载方法

  + 第一个将基本数据类型（int/long/ulong/double/string）作为输入参数。
  + 第二个将Dictionary<string, T>作为输入参数，其中字典的key一般表示的是pin name，字典的value表示的是测量值其数据类型可以是（int/long/ulong/double/string）。

Testmethod example

[复制](# "复制")

```
public class OS_Test : TestMethod
{
    public string Pins { set; get; }
    public double IForce { set; get; }
    public double VLimit { set; get; }
    public ParametricTestDescriptor ptdVoltage { set; get; }

    public override void Execute()
    { //Mandatory

        //*****OS Test*****
        var pinsDcvi = SemiContext.DCVI(Pins);

        // Converting the IForce value from mA to A
        double iForceValue = IForce / 1000;

        //Forcing voltage
        pinsDcvi.IForce(iForceValue, 0.32, 0.5, sourceDelay: 0);

        //Current Measurement
        Dictionary<string, double> pinsVoltageMeasurements = pinsDcvi.IForceVMeasure(0, 6);

        //Evaluation of the measured results
        ptdVoltage.Evaluate(pinsVoltageMeasurements);
        //do required post processing here
    }
}
```

|  | 4.2.FrontDialog API |

4.2.Dialog API

* 在Test Method中可以通过 IFrontDialog 来自定义用户需要定制化的各种弹窗

Testmethod example

[复制](# "复制")

```
using Guwave.OneTest.TestMethod;

namespace UserDefinedTestMethods
{
    [DefinedTestMethod("This a show dialog example")]
    public class Example_Dialog : TestMethod
    {
        public IFrontDialog MyDialog { get; set; }

        public override void Execute()
        {
            Log(">>>> Example_Dialog - execute...");
            // 常见的确认对话框
            MyDialog.Description = "请确认XXX线已连接完成！";
            MyDialog.Show();

            // 输入参数弹框
            MyDialog.Description = "请输入用户姓名：";
            MyDialog.AddInputForm("Name", "姓名", "张三");
            DialogResult result = MyDialog.ShowForResult();
            Log($">>>> input data: {result.InputData}");

            // 多表单定制化输入用法
            MyDialog.RemoveForm("Name");      // 删除前面设置的输入姓名表单
            MyDialog.Description = "请设置当前环境温度：";
            MyDialog.AddNumberSelect("Temp", "温度", 32, "度");
            List<string> itemList = new List<string>(){"30%", "50%", "65%", "75%"};
            MyDialog.AddSelectForm("XXX", "湿度", itemList, "50%");
            MyDialog.AddButton("Cancel", "取消");
            result = MyDialog.ShowForResult();
            if(result.ButtonKey == "OK")
                Log("用户选择了确认温度：" + result.InputData["Temp"]);
            else
                Log("用户放弃了输入温度");
            MyDialog.RemoveButton("Cancel");    // 删除取消按钮
            MyDialog.ClearForms();

            // 单选框组定制化输入
            MyDialog.Description = "请选择当前连接的YY类型：";
            MyDialog.AddRadioGroup("YYY", "类型", new List<string>(){"Local", "Remote"}, "Remote");
            MyDialog.AddSelectForm("FFF", "性别", new List<string>(){"女:0", "男:1", "未知:2"}, "1");

            Log(">>>> 用户选择类型：" + result.InputData["YYY"]);
            Log(">>>> 用户选择性别：" + result.InputData["FFF"]);
            MyDialog.RemoveForm("YYY");

            // 复选框
            MyDialog.Description = "请选择当前连接的ZZ类型：";
            itemList = new List<string>(){"XXX", "YYY", "ZZZ"};
            MyDialog.AddCheckGroup("ZZZ", "选择类型", itemList, "ZZZ");
            result = MyDialog.ShowForResult();

            Log(">>>> Example_Dialog - done...");

## 4.5.InstrumentAccessor API

|  |  |
| --- | --- |
|  | 4.5.InstrumentAccessor API |

4.5.InstrumentAccessor API

* 在Test Method中可以通过InstrumentAccessor对指定的仪表关闭或初始化，常用于解决仪表OLP问题，当仪表发生OLP异常时可以在TestMthoed中对仪表做重置然后重新初始化。

Testmethod example

[复制](# "复制")

```
using Guwave.OneTest.TestMethod;
using System;

namespace UserDefinedTestMethods
{
    [DefinedTestMethod("This is a Common OS Test")]
    public class CurrentTest : TestMethod
    {
        [InputParam(DefaultValue = "VCC1", Description = "Pin or PinGroup Name", OptionValue = "VCC1,VCC2")]
        public string Pins { get; set;  }

        [InputParam(DefaultValue = 3.3, Description = "Force Voltage Level (V)")]
        public double VForce { get; set;  }
        [InputParam(DefaultValue = 0.1, Description = "Current Limit (mA)")]
        public double ILimit { get; set;  }

        public ParametricTestDescriptor ptdCurrent { get; set;  }

        public override void Execute()
        { 
            var pinsDcvi = SemiContext.DCVI(Pins);
            //Forcing voltage
            pinsDcvi.VForce(VForce);
            pinsDcvi.IForce(0.02);
            //Current Measurement
            Dictionary<string, Double> pinsCurrentMeasurements = pinsDcvi.VForceIMeasure(0.5);
            ptdCurrent.Evaluate(pinsCurrentMeasurements);
            //通过PinName获取所连接仪表的名称
            string instName = this.PinmapAccessor.GetInstrumentName("VCC1");
            this.Log("InstrumentName === " + instName);

            pinsDcvi.ResetDevice();  //对仪表硬件进行重置
            this.InstrumentAccessor.CloseByName(instName);  //断开仪表连接
            Thread.Sleep(3000);
            this.InstrumentAccessor.InitialByName(instName); //初始化仪表连接

            pinsDcvi = SemiContext.DCVI(Pins);
            pinsCurrentMeasurements = pinsDcvi.VForceIMeasure(0.5);
            ptdCurrent.Evaluate(pinsCurrentMeasurements);

            pinsDcvi.ResetDevice();  //对仪表硬件进行重置
            this.InstrumentAccessor.CloseByModel("NI4145");    //断开同型号所有仪表的连接
            Thread.Sleep(3000);
            this.InstrumentAccessor.InitialByModel("NI4145");  //初始化同型号所有仪表

            pinsDcvi = SemiContext.DCVI(Pins);
            pinsCurrentMeasurements = pinsDcvi.VForceIMeasure(0.5);
            ptdCurrent.Evaluate(pinsCurrentMeasurements);
        }
    }
}

4.7.RegisterAccesser API

* Apply/Verify寄存器组的定义是：对寄存器组中的只读寄存器进行读操作， 对只写寄存器进行只写寄存器， 对于可读可写的寄存器先写后读。

namespace UserDefinedTestMethods
